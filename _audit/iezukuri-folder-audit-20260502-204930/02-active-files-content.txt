$attachment_url = static function ($attachment_id, $size = 'full') {
    $attachment_id = absint($attachment_id);
    if (!$attachment_id) {
        return '';
    }

    $mime = get_post_mime_type($attachment_id);
    if (strpos((string) $mime, 'image/') === 0) {
        $url = wp_get_attachment_image_url($attachment_id, $size);
        return $url ? $url : '';
    }

    $url = wp_get_attachment_url($attachment_id);
    return $url ? $url : '';
};

$page_url = static function ($slugs, $fallback = '/') {
    if (function_exists('naigai_customhome_find_page_url')) {
        return naigai_customhome_find_page_url($slugs, $fallback);
    }

    foreach ((array) $slugs as $slug) {
        $page = get_page_by_path($slug);
        if ($page) {
            return get_permalink($page->ID);
        }
    }

    return home_url($fallback);
};

/**
 * =========================================================
 * Iezukuri body classes
 *
 * 既存CSSが .hub-customhome-subpage / --{slug} を前提にしているため、
 * /iezukuri/ サブページに必要な body class を付与する。
 * =========================================================
 */
add_filter('body_class', function ($classes) {
    if (!is_page()) {
        return $classes;
    }

    $post = get_post();
    if (!$post) {
        return $classes;
    }

    $is_iezukuri_child = false;

    foreach (get_post_ancestors($post) as $ancestor_id) {
        if (get_post_field('post_name', $ancestor_id) === 'iezukuri') {
            $is_iezukuri_child = true;
            break;
        }
    }

    if ($is_iezukuri_child) {
        $classes[] = 'hub-customhome-subpage';
        $classes[] = 'hub-customhome-subpage--' . sanitize_html_class($post->post_name);
    }

    if ($post->post_name === 'iezukuri') {
        $classes[] = 'hub-customhome-page';
    }

    return array_values(array_unique($classes));
});

if (!function_exists('naigai_iezukuri_is_context_page')) {
    function naigai_iezukuri_is_context_page()
    {
        if (is_admin()) {
            return false;
        }

        if (is_page_template(array(
            'template-construction-hub.php',
            'page-construction-hub-sub.php',
            'iezukuri/templates/template-iezukuri.php',
            'iezukuri/templates/template-iezukuri-subpage.php',
        ))) {
            return true;
        }

        if (!is_page()) {
            return false;
        }

        $post = get_post();

        if (!$post) {
            return false;
        }

        if ($post->post_name === 'iezukuri') {
            return true;
        }

        foreach (get_post_ancestors($post) as $ancestor_id) {
            if (get_post_field('post_name', $ancestor_id) === 'iezukuri') {
                return true;
            }
        }

        return false;
    }
}

add_action('wp_enqueue_scripts', function () {
    if (!naigai_iezukuri_is_context_page()) {
        return;
    }

    $theme_dir = get_template_directory();
    $theme_uri = get_template_directory_uri();

    $main_css = '/iezukuri/css/iezukuri.css';
    $sub_css  = '/iezukuri/css/iezukuri-subpage.css';

    if (file_exists($theme_dir . $main_css)) {
        wp_enqueue_style(
            'naigai-iezukuri',
            $theme_uri . $main_css,
            array(),
            filemtime($theme_dir . $main_css)
        );
    }

    if (!is_page('iezukuri') && file_exists($theme_dir . $sub_css)) {
        wp_enqueue_style(
            'naigai-iezukuri-subpage',
            $theme_uri . $sub_css,
            array('naigai-iezukuri'),
            filemtime($theme_dir . $sub_css)
        );
    }
$post = get_post();
    $slug = $post ? $post->post_name : '';

    $page_css = $slug ? '/iezukuri/css/pages/' . $slug . '.css' : '';
    if ($page_css && file_exists($theme_dir . $page_css)) {
        wp_enqueue_style(
            'naigai-iezukuri-page-' . sanitize_key($slug),
            $theme_uri . $page_css,
            array('naigai-iezukuri-subpage', 'naigai-iezukuri-ch-hero'),
            filemtime($theme_dir . $page_css)
        );
    }

    $js_files = array(
        '/iezukuri/js/iezukuri.js',
        '/iezukuri/js/customhome-nav.js',
        '/iezukuri/js/ch-hero.js',
    );

    foreach ($js_files as $js) {
        if (file_exists($theme_dir . $js)) {
            wp_enqueue_script(
                'naigai-' . sanitize_key(basename($js, '.js')),
                $theme_uri . $js,
                array(),
                filemtime($theme_dir . $js),
                true
            );
        }
    }
}, 40);
